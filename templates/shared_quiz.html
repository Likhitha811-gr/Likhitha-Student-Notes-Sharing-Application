@app.route("/shared-quiz/", methods=["GET", "POST"]) def shared_quiz(token): conn = get_db_connection() cursor = conn.cursor() cursor.execute(""" SELECT quiz_json FROM shared_quizzes WHERE token = %s """, (token,)) row = cursor.fetchone() cursor.close() conn.close() if not row: abort(404) quiz_questions = json.loads(row[0]) score = None total = len(quiz_questions) submitted = request.method == "POST" if submitted: score = 0 for question in quiz_questions: selected = request.form.get( f"q{question['number']}" ) selected_index = None if selected is not None and selected.isdigit(): selected_index = int(selected) correct_index = int(question["answer"]) question["selected"] = selected_index question["is_correct"] = ( selected_index == correct_index ) if question["is_correct"]: score += 1 return render_template( "shared_quiz.html", quiz_questions=quiz_questions, score=score, total=total, submitted=submitted )
